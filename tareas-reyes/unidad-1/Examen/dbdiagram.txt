Table Recinto {
  id_recinto int [pk, not null] // PK
  nombre varchar [not null]
  capacidad int [not null, check: `capacidad > 0`]
  disponibilidad int [not null, check: `disponibilidad >= 0 AND disponibilidad <= capacidad`]
}

Table Especie {
  id_especie int [pk, not null] // PK
  nombre varchar [not null]
}

Table Animal {
  id_animal int [pk, not null] // PK
  id_especie int [not null] // FK
  nombre varchar [not null]
}

Table Evaluacion_Medica {
  id_evaluacion_medica int [pk, not null] // PK
  id_animal int [not null] // FK
  diagnostico varchar [not null]
  tratamiento varchar [not null]
  estado_animal varchar [not null] // 'Critico', 'Estable', etc
  fecha timestamp [not null]
}

Table Movimiento {
  id_movimiento int [pk, not null] // PK
  id_animal_id int [not null] // FK
  id_recinto_destino int [not null] // FK
  motivo varchar [not null]
  fecha timestamp [not null]
}
Ref: Evaluacion_Medica.id_animal > Animal.id_animal
Ref: Movimiento.id_recinto_destino > Recinto.id_recinto
Ref: Animal.id_animal < Movimiento.id_animal_id
Ref: Animal.id_especie > Especie.id_especie

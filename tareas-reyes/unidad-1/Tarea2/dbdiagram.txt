Table cliente {
  DNI varchar [primary key] 
  nombre varchar
  telefono varchar
  email varchar
}

Table mascota {
  id integer [primary key]
  nombre varchar
  especie varchar
  raza varchar
  fecha_de_nacimiento date
  DNI_dueño varchar [not null]
}

Table veterinario {
  DNI varchar [primary key] 
  nombre varchar
  especialidad_medica varchar
  telefono varchar
}

Table atencion_medica {
  id integer [primary key]
  DNI_veterinario varchar [not null]
  id_mascota integer [not null]
  fecha_y_hora timestamp
  diagnostico varchar
  costo decimal
}

Table medicamento {
  id integer [primary key]
  nombre varchar
  fabricante varchar
  costo decimal
}

Table prescripcion_medica {
  id integer [primary key]
  id_medicamento integer [not null]
  id_atencion_medica integer [not null]
  indicaciones varchar
}
